{{/* nav/language-toggle Language switcher for bilingual sites: a single direct link to the other language, without a dropdown. Alternative to nav/language, which lists every language in a dropdown menu. Not called by the theme: to use it, override a layout (header/panel/content.html for instance) and call this partial in place of nav/language. Context: the page. Reads .Page.Translations and hugo.Sites, one site per language; a language the page has no translation in links to its home page. Caveat: with three languages or more, only the first alternative language is rendered, as the range stops at the first match. Renders nothing on a monolingual site. */}} {{- if hugo.IsMultilingual }} {{- $pageTrans := .Page.Translations -}} {{- $current := partial "func/GetLanguage.html" -}}
  • {{- range hugo.Sites -}} {{- $language := partial "func/GetLanguage.html" .Language -}} {{- if ne $language.name $current.name -}} {{- $url := .Home.Permalink -}} {{- range $pageTrans -}} {{- if eq (partial "func/GetLanguage.html" .Language).name $language.name -}} {{- $url = .Permalink -}} {{- end -}} {{- end -}} {{- with $language -}} {{ upper .name }} {{- end -}} {{- break -}} {{ end -}} {{- end -}}
{{ end -}}
