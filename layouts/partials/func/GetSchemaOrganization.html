{{/* GetSchemaOrganization Build the schema.org node for the organization behind the site, or for one of its offices. Edited in the CMS as data/schema.yml (Config → Structured data), under the `organization` key, in the very same shape a place uses so both feed this partial unchanged. The file is translated, and func/GetSchemaData.html resolves it for the current language — a locale only has to carry what it changes: organization: name: Acme type: Organization # any Organization subtype, see # func/GetSchemaTypes.html image: src: '/assets/images/logo.svg' address: street: 1 rue de l'Exemple zipcode: '75001' city: Paris country: France coordinates: lat: 48.8566 lng: 2.3522 contact: email: … phone: … fax: … socials: - platform: linkedin url: … Passing a place turns the node into that office: it keeps the logo the whole firm shares, overrides whatever the place describes for itself, takes its own @id and points back at the head office through parentOrganization. That is how a person attached to a place ends up working for their office rather than for the firm as a whole. A place only qualifies as an office when its schema.type names an organization subtype — the places section is generic, and a venue or a landmark is a location, not an organization. Otherwise it stays a place: see func/GetSchemaPlace.html, and func/GetSchemaEntity.html which picks between the two. The key is nested rather than a top level `type`, which Hugo reserves in front matter to select the content type. Params: {map} optional compact (bool) return only what another node needs to reference this one: @type, @id, name, url and the logo — which Google reads on a publisher or a hiringOrganization. place (page) a /places/ page to build a branch office from. name (string) fallback name when the data file has none. Defaults to the site title. Returns: map, the schema.org node. */}} {{- $compact := false -}} {{- $place := false -}} {{- $name := site.Title -}} {{- with . -}} {{- $compact = .compact | default false -}} {{- $place = .place | default false -}} {{- with .name -}}{{- $name = . -}}{{- end -}} {{- end -}} {{- $datas := partial "func/GetSchemaData.html" "organization" | default dict -}} {{- with $datas.name -}}{{- $name = . -}}{{- end -}} {{- $type := $datas.type | default "Organization" -}} {{- $headName := $name -}} {{- $headType := $type -}} {{- $headId := printf "%s#organization" site.BaseURL -}} {{- $id := $headId -}} {{- $url := site.BaseURL -}} {{- /* The logo belongs to the firm, wherever the office is. It carries an @id of its own so every copy of it — the full node, each compact reference — merges back into a single image. */ -}} {{- $logo := false -}} {{- with partial "func/GetSchemaImage.html" $datas.image -}} {{- $logo = merge . (dict "@id" (printf "%s#logo" site.BaseURL)) -}} {{- end -}} {{- /* An office of the firm: its own identity, and its own type when it declares one — a group can run a clinic and a shop. A type from the place branch is no organization, the firm's one stands then. */ -}} {{- with $place -}} {{- /* An office kept out of the build (render: never) has neither: its node goes anonymous rather than claim the firm's @id. */ -}} {{- $id = cond (ne .Permalink "") (printf "%s#organization" .Permalink) "" -}} {{- $url = .Permalink -}} {{- $name = .Title -}} {{- with .Params.schema -}} {{- with .type -}} {{- if in (partial "func/GetSchemaTypes.html").organization . -}} {{- $type = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $organization := dict "@type" $type "@id" $id "name" (partial "func/PrepareData.html" $name) "url" $url -}} {{- with $logo -}} {{- $organization = merge $organization (dict "logo" .) -}} {{- end -}} {{- if not $compact -}} {{- /* The description is the site one, which data/seo.yml already holds per language, so data/schema.yml needs no prose of its own. An office overrides it from its own page, which is translatable content. */ -}} {{- $description := partial "func/GetSeoData.html" "description" -}} {{- /* What an office describes for itself wins; the firm fills in the rest. The address is taken whole — half an office's address completed with the head office's would point nowhere — where a contact is completed field by field: the switchboard still reaches an office that has no line of its own. */ -}} {{- $address := $datas.address -}} {{- $contact := $datas.contact | default dict -}} {{- $image := $logo -}} {{- with $place -}} {{- with .Description -}}{{- $description = . -}}{{- end -}} {{- with .Params.address -}}{{- $address = . -}}{{- end -}} {{- with .Params.contact -}} {{- $contact = partial "func/MergeDefined.html" (dict "base" $contact "overlay" .) -}} {{- end -}} {{- with partial "func/GetSchemaImage.html" .Params.image -}}{{- $image = . -}}{{- end -}} {{- end -}} {{- /* Set once: merge recurses into a map already there, and the office photo would inherit the @id of the logo. */ -}} {{- with $image -}} {{- $organization = merge $organization (dict "image" .) -}} {{- end -}} {{- with $description -}} {{- $organization = merge $organization (dict "description" (partial "func/PrepareData.html" .)) -}} {{- end -}} {{- with partial "func/GetSchemaAddress.html" $address -}} {{- $organization = merge $organization (dict "address" .) -}} {{- end -}} {{- with partial "func/GetSchemaGeo.html" $address -}} {{- $organization = merge $organization (dict "geo" .) -}} {{- end -}} {{- $organization = merge $organization (partial "func/GetSchemaContact.html" $contact) -}} {{- with $place -}} {{- $parent := dict "@type" $headType "@id" $headId "name" (partial "func/PrepareData.html" $headName) "url" site.BaseURL -}} {{- with $logo -}}{{- $parent = merge $parent (dict "logo" .) -}}{{- end -}} {{- $organization = merge $organization (dict "parentOrganization" $parent) -}} {{- end -}} {{- end -}} {{- return $organization -}}
