{{/* GetImageResizeSrc The URL of an image resized to a size, in a given format, whatever its provider. Used by the lightbox (commons/image-lightbox), whose fallback and srcset are built from it. - Uploadcare: -/resize/ then -/format/, appended to /UUID/ (the file name stripped, else 404) - Cloudinary: w_/h_ then f_ between /upload/ and the file - Unsplash: Imgix w/h, fit=max then fm (auto=format without a type) - Hugo: the asset resized, the format as a Resize option Param: dict { image (URL or asset path), size ("WxH", "Wx"), type (format, e.g. "webp", optional) } Returns: string, false when the asset is not found */}} {{- $size := .size -}} {{- $type := .type | default "" -}} {{- $image_src := false -}} {{- with .image -}} {{- if or (in . "ucarecd.net") (in . "ucarecdn.com") -}} {{- $file := (partial "func/GetUploadcareFile.html" .).file -}} {{- $image_src = printf "%s-/resize/%s/" $file $size -}} {{- with $type -}} {{- $image_src = printf "%s-/format/%s/" $image_src . -}} {{- end -}} {{- else if in . "cloudinary.com" -}} {{- $cloudinary := partial "func/GetCloudinaryParams.html" (dict "file" .) -}} {{- $params := partial "func/GetCloudinarySize.html" $size -}} {{- with $type -}} {{- $params = $params | append (printf "f_%s" .) -}} {{- end -}} {{- $image_src = printf "%s%s/%s" $cloudinary.path (delimit $params ",") $cloudinary.file -}} {{- else if in . "images.unsplash.com" -}} {{- $parts := split $size "x" -}} {{- $image_src = partial "func/GetUnsplashSrc.html" (dict "unsplash" (partial "func/GetUnsplashParams.html" (dict "file" .)) "width" (int (index $parts 0 | default 0)) "height" (int (index $parts 1 | default 0)) "format" $type ) -}} {{- else -}} {{- with resources.GetMatch . -}} {{- $image_src = (.Resize (printf "%s %s" $size $type)).RelPermalink -}} {{- end -}} {{- end -}} {{- end -}} {{- return $image_src -}}
