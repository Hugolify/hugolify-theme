{{/* GetSchemaVideo Build the schema.org VideoObject of a video the page plays: the video block, the video shortcode. Only a video someone watches gets one, a video with controls: an autoplaying loop without them is a moving background. Google shows its video features for a page whose main purpose is the video, a watch page; elsewhere the node still describes the video to search engines. Google requires a name, a thumbnail and an upload date: without one of them the node is dropped rather than published invalid. name the title the caller gives (the heading of the block), the legend of the video, the title of the page description the text the caller gives, the description of the page, its seo.description first thumbnailUrl the poster, or the frame Cloudinary draws from a video it hosts, its extension turned into .jpg — Uploadcare and a local file have none to give uploadDate the publication date of the page, its last change otherwise contentUrl the file, the MP4 first duration read from the MP4 itself, see func/GetVideoDuration.html transcript the transcription Params: {map} video (map) the video object: mp4, webm, poster, legend, transcription, controls… page (page) optional, the page it plays on. Defaults to the page being rendered. name (string) optional, the title of the video. description (string) optional, its description, Markdown allowed. Returns: map, the VideoObject node — or false. */}} {{- $video := .video | default dict -}} {{- $page := .page | default page -}} {{- $node := false -}} {{- if $video.controls -}} {{- with or $video.mp4 $video.webm -}} {{- $src := . -}} {{- $url := . -}} {{- if hasPrefix $url "//" -}} {{- $url = printf "https:%s" $url -}} {{- else if not (hasPrefix $url "http") -}} {{- $url = absURL $url -}} {{- end -}} {{- $name := partial "func/PrepareData.html" (or $.name $video.legend (partial "func/GetRealContext.html" $page).Title) -}} {{- $thumbnail := "" -}} {{- with partial "func/GetSchemaImage.html" $video.poster -}} {{- $thumbnail = .url -}} {{- else -}} {{- if findRE `^https://res\.cloudinary\.com/[^/]+/video/upload/` $url -}} {{- $thumbnail = printf "%s.jpg" (replaceRE `\.[A-Za-z0-9]+$` "" $url) -}} {{- end -}} {{- end -}} {{- $date := or (partial "func/GetSchemaDate.html" $page.PublishDate) (partial "func/GetSchemaDate.html" $page.Lastmod) -}} {{- if and $name $thumbnail $date -}} {{- $node = dict "@type" "VideoObject" "@id" (printf "%s#video-%s" $page.Permalink (substr (md5 $url) 0 8)) "name" $name "thumbnailUrl" $thumbnail "uploadDate" $date "contentUrl" $url "encodingFormat" (cond (eq $src $video.mp4) "video/mp4" "video/webm") "inLanguage" (partial "func/GetLanguage.html" $page.Language).locale -}} {{- /* The page ones the way func/GetSchemaWebPage.html reads them, seo.description first. */ -}} {{- $description := $.description -}} {{- if not $description -}} {{- $description = $page.Description -}} {{- with $page.Params.seo -}}{{- with .description -}}{{- $description = . -}}{{- end -}}{{- end -}} {{- end -}} {{- with $description -}} {{- $node = merge $node (dict "description" (partial "func/PrepareData.html" (markdownify .))) -}} {{- end -}} {{- /* The duration the MP4 holds, in ISO 8601: PT1M30S. */ -}} {{- with $video.mp4 -}} {{- with partial "func/GetVideoDuration.html" . -}} {{- $iso := "PT" -}} {{- with div . 3600 -}}{{- $iso = printf "%s%dH" $iso . -}}{{- end -}} {{- with div (mod . 3600) 60 -}}{{- $iso = printf "%s%dM" $iso . -}}{{- end -}} {{- if or (mod . 60) (eq $iso "PT") -}}{{- $iso = printf "%s%dS" $iso (mod . 60) -}}{{- end -}} {{- $node = merge $node (dict "duration" $iso) -}} {{- end -}} {{- end -}} {{- with $video.transcription -}} {{- $node = merge $node (dict "transcript" (partial "func/PrepareData.html" .)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $node -}}
