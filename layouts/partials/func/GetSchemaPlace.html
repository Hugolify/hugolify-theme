{{/* GetSchemaPlace Build the schema.org node for a place that is a location. A place is a location by default — a venue, a landmark, an address people go to — and says which kind through schema.type. Declaring an organization subtype there makes it an office instead, and func/GetSchemaOrganization.html describes it. The two nodes take different @id suffixes so both can coexist: an office is also the location its people work at, or a job is offered at. Only properties Place actually defines are emitted: email and social profiles belong to the organization or the person, not to a location. Params: (page) a /places/ page, or the taxonomy term standing in for one when GetRealPlace found none, or the page-like map of a place kept in data, which has no URL to give. {map} to pass options: page (page) as above. compact (bool) return only what another node needs to reference this one: @type, @id, name and url. Returns: map, the Place node. */}} {{- $place := . -}} {{- $compact := false -}} {{- if and (reflect.IsMap .) (isset . "page") -}} {{- $place = .page -}} {{- $compact = .compact | default false -}} {{- end -}} {{- $isPage := not (reflect.IsMap $place) -}} {{- $type := "Place" -}} {{- with $place.Params.schema -}} {{- with .type -}} {{- if in (partial "func/GetSchemaTypes.html").place . -}} {{- $type = . -}} {{- end -}} {{- end -}} {{- end -}} {{- $node := dict "@type" $type "name" (partial "func/PrepareData.html" $place.Title) -}} {{- /* A section kept out of the build (render: never) has no URL either. */ -}} {{- if $isPage -}} {{- with $place.Permalink -}} {{- $node = merge $node (dict "@id" (printf "%s#place" .) "url" .) -}} {{- end -}} {{- end -}} {{- if not $compact -}} {{- with $place.Description -}} {{- $node = merge $node (dict "description" (partial "func/PrepareData.html" .)) -}} {{- end -}} {{- with partial "func/GetSchemaImage.html" $place.Params.image -}} {{- $node = merge $node (dict "image" .) -}} {{- end -}} {{- with partial "func/GetSchemaAddress.html" $place.Params.address -}} {{- $node = merge $node (dict "address" .) -}} {{- end -}} {{- with partial "func/GetSchemaGeo.html" $place.Params.address -}} {{- $node = merge $node (dict "geo" .) -}} {{- end -}} {{- with $place.Params.contact -}} {{- with .phone -}}{{- $node = merge $node (dict "telephone" (string .)) -}}{{- end -}} {{- with .fax -}}{{- $node = merge $node (dict "faxNumber" (string .)) -}}{{- end -}} {{- end -}} {{- end -}} {{- return $node -}}
