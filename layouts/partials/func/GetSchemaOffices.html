{{/* GetSchemaOffices Build the schema.org nodes of the offices of the organization, for the page that lists them: its contact page. The contact page is the one data/schema.yml names, contact_page next to organization, or the /contact page when it names none. Params: (page) the page. "section" "places") -}} {{- if $real.File -}} {{- $offices = $offices | append (partial "func/GetSchemaOrganization.html" (dict "place" $real "page" $page)) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $offices -}}
