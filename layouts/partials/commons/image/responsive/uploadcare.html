{{/* Responsive image (srcset width descriptors) — Uploadcare. Used by commons/image.html when the caller gives `sizes` or `widths`, and by default for a lazy image: one srcset of widths (w descriptors) plus `sizes` ("auto" when lazy), the browser picks the file for the rendered width and the screen density. The fixed-size variant (eager, src_mobile, small images) is in ../default/. Transformations are generated on the fly (no build cost), so a dense ladder is fine. Retina is handled by the browser via DPR; variants larger than the source are skipped. */}} {{- $quality := site.Params.image.quality | default "normal" -}} {{/* crop alignment (lowercased); crop_ratio (h/w) enables cropped variants via scale_crop. */}} {{- $crop := lower (.crop | default "center") -}} {{- $crop_ratio := .crop_ratio -}} {{/* File name stripped for the transformations, SVG told by its extension */}} {{- $uploadcare := partial "func/GetUploadcareFile.html" .file -}} {{- $file := $uploadcare.file -}} {{- $is_svg := $uploadcare.is_svg -}} {{- $image_desktop := $file -}} {{- $image_width := false -}} {{- $image_height := false -}} {{- $srcset := slice -}} {{/* Intrinsic size: caps the ladder and feeds the aspect ratio, an SVG's too (no CLS). */}} {{- $size := partial "func/GetRemoteImageSize.html" (dict "url" (printf "%s-/json/" $file) "provider" "uploadcare") -}} {{- $image_width = $size.width -}} {{- $image_height = $size.height -}} {{/* An URL without file name (/UUID/) has no extension: the format reported by Uploadcare tells */}} {{- $is_svg = or $is_svg (eq $size.format "svg") -}} {{/* SVG is resolution-independent: served as-is, no srcset nor transformation. */}} {{- if not $is_svg -}} {{/* format/auto: Uploadcare negotiates AVIF/WebP/original per browser via the Accept header, so a single srcset suffices — no hand-built webp/avif needed. */}} {{- $q := printf "-/format/auto/-/quality/%s/" $quality -}} {{/* The ladder, capped by the source width (see func/GetSrcsetSizes) */}} {{- $sizes := partial "func/GetSrcsetSizes.html" (dict "widths" .widths "max" $image_width "crop_ratio" $crop_ratio "provider" "uploadcare") -}} {{- $fallback := false -}} {{- $fallback_w := false -}} {{- $fallback_h := false -}} {{- range $sizes -}} {{- $w := .width -}} {{- $h := .height -}} {{- if $crop_ratio -}} {{/* Crop: scale_crop to w×h with the chosen alignment. */}} {{- $srcset = $srcset | append (printf "%s-/scale_crop/%dx%d/%s/%s %dw" $file $w $h $crop $q $w) -}} {{- $fallback = printf "%s-/scale_crop/%dx%d/%s/%s" $file $w $h $crop $q -}} {{- $fallback_w = $w -}} {{- $fallback_h = $h -}} {{- else -}} {{- $srcset = $srcset | append (printf "%s-/resize/%dx/%s %dw" $file $w $q $w) -}} {{- $fallback = printf "%s-/resize/%dx/%s" $file $w $q -}} {{- end -}} {{- end -}} {{/* Not `default`: it takes a boolean, false included, as a set value */}} {{- with $fallback -}} {{- $image_desktop = . -}} {{- end -}} {{/* For a crop, the dimensions are the fallback variant's computed crop size. */}} {{- if $crop_ratio -}} {{- $image_width = $fallback_w -}} {{- $image_height = $fallback_h -}} {{- end -}} {{- end -}} {{- return dict "image_desktop" $image_desktop "image_width" $image_width "image_height" $image_height "srcset" $srcset -}}
