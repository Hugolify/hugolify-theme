{{/* Default image (fixed sizes, art direction) — Uploadcare. Used by commons/image.html when the responsive mode is off: no `sizes` nor `widths` given, and an eager image (lazy: false, e.g. the hero), a distinct mobile source (src_mobile) or a small image whose requested width is below image.responsive_min_width (320 by default). Each requested size (desktop, mobile) gives its own file, retina comes as 2x/3x density variants, the mobile one is served under (max-width: 576px). The srcset + sizes variant (browser picks the width) is in ../responsive/. Uploadcare: transformations on the fly; one desktop size, the mobile one in 1x/2x/3x; an SVG is served as-is. */}} {{/* Get params */}} {{- $file := .file -}} {{- $file_mobile := .file_mobile -}} {{- $desktop_size := .desktop_size -}} {{- $mobile_size := .mobile_size -}} {{- $mobile_size_2x := .mobile_size_2x -}} {{- $mobile_size_3x := .mobile_size_3x -}} {{- $crop := lower .crop -}} {{- $quality := site.Params.image.quality | default "normal" -}} {{/* SVG told by its extension, read before the file name is stripped below */}} {{- $is_svg := strings.HasSuffix (lower $file) ".svg" -}} {{/* Strip filename from Uploadcare URL if present (e.g., /UUID/filename.ext -> /UUID/) */}} {{- $stripFilenamePattern := `(/[a-f0-9-]+)/[^/]+\.[a-zA-Z]+/?$` -}} {{- $file = replaceRE $stripFilenamePattern "$1/" $file -}} {{- if $file_mobile -}} {{- $file_mobile = replaceRE $stripFilenamePattern "$1/" $file_mobile -}} {{- end -}} {{/* Return params */}} {{- $image_desktop := $file -}} {{- $image_mobile := false -}} {{- $image_mobile_2x := false -}} {{- $image_mobile_3x := false -}} {{- $image_width := false -}} {{- $image_height := false -}} {{/* Get image dataset */}} {{- $url := printf "%s-/json/" $file }} {{- with try (resources.GetRemote $url) -}} {{- with .Err -}} {{ errorf "%s" . }} {{- else with .Value -}} {{- $data := . | transform.Unmarshal -}} {{- with $data -}} {{- $image_width = index . "width" -}} {{- $image_height = index . "height" -}} {{/* An URL without file name (/UUID/) has no extension: the format reported by Uploadcare tells */}} {{- with index . "format" -}} {{- $is_svg = or $is_svg (eq (lower .) "svg") -}} {{- end -}} {{- end -}} {{- else -}} {{ errorf "Unable to get remote resource: %s" $url }} {{- end -}} {{- end -}} {{/* Process if not a SVG image */}} {{- if not $is_svg }} {{- $quality := printf "-/quality/%s/" $quality -}} {{/* Return desktop image */}} {{- with $desktop_size -}} {{- $params := printf "-/resize/%s/%s" . $quality -}} {{- if partial "func/isCrop" . -}} {{- $params = printf "-/scale_crop/%s/%s/%s" . $crop $quality -}} {{- end -}} {{- $image_desktop = printf "%s%s" $file $params -}} {{/* Get modificated image dataset */}} {{- $url = printf "%s-/resize/%s/-/json/" $file . }} {{- with try (resources.GetRemote $url) -}} {{- with .Err -}} {{- errorf "Error, GetRemote: %s" . -}} {{- else with .Value -}} {{- $data := . | transform.Unmarshal -}} {{- with $data -}} {{- $image_width = index . "width" -}} {{- $image_height = index . "height" -}} {{- end -}} {{- else -}} {{- errorf "Error, unable to retrieve remote resource for: %s" $file -}} {{- end -}} {{- end -}} {{- end -}} {{/* Return mobile image */}} {{- if $mobile_size -}} {{- $params := printf "-/resize/%s/%s" $mobile_size $quality -}} {{- $params_2x := printf "-/resize/%s/%s" $mobile_size_2x $quality -}} {{- $params_3x := printf "-/resize/%s/%s" $mobile_size_3x $quality -}} {{- if partial "func/isCrop" $mobile_size -}} {{- $params = printf "-/scale_crop/%s/%s/%s" $mobile_size $crop $quality -}} {{- $params_2x = printf "-/scale_crop/%s/%s/%s" $mobile_size_2x $crop $quality -}} {{- $params_3x = printf "-/scale_crop/%s/%s/%s" $mobile_size_3x $crop $quality -}} {{- end -}} {{- $image_mobile = printf "%s%s" $file_mobile $params -}} {{- $image_mobile_2x = printf "%s%s" $file_mobile $params_2x -}} {{- $image_mobile_3x = printf "%s%s" $file_mobile $params_3x -}} {{- end -}} {{- end -}} {{- return dict "image_desktop" $image_desktop "image_mobile" $image_mobile "image_mobile_2x" $image_mobile_2x "image_mobile_3x" $image_mobile_3x "image_width" $image_width "image_height" $image_height -}}
