{{/* Default image (fixed sizes, art direction) — Cloudinary. Used by commons/image.html when the responsive mode is off: no `sizes` nor `widths` given, and an eager image (lazy: false, e.g. the hero), a distinct mobile source (src_mobile) or a small image whose requested width is below image.responsive_min_width (320 by default). Each requested size (desktop, mobile) gives its own file, retina comes as 2x/3x density variants, the mobile one is served under (max-width: 576px). The srcset + sizes variant (browser picks the width) is in ../responsive/. Cloudinary: transformations on the fly, f_auto negotiates the format; desktop retina as a dpr_2.0 variant (image.retina.desktop). */}} {{/* Get params */}} {{- $is_desktop_retina := site.Params.image.retina.desktop | default true -}} {{- $desktop_size := .desktop_size -}} {{- $mobile_size := .mobile_size -}} {{/* /upload//, f_auto + quality, crop gravity */}} {{- $cloudinary := partial "func/GetCloudinaryParams.html" (dict "file" .file "crop" .crop) -}} {{- $pathFile := $cloudinary.path -}} {{- $imageFile := $cloudinary.file -}} {{- $crop := $cloudinary.gravity -}} {{/* Return params */}} {{- $image_desktop := .file -}} {{- $image_desktop_2x := false -}} {{- $image_mobile := false -}} {{- $image_mobile_2x := false -}} {{- $image_mobile_3x := false -}} {{/* Intrinsic size, an SVG read as a PNG */}} {{- $info := cond $cloudinary.is_svg "fl_getinfo/f_png/" "fl_getinfo/" -}} {{- $size := partial "func/GetRemoteImageSize.html" (dict "url" (print $pathFile $info $imageFile) "provider" "cloudinary" "strict" true) -}} {{- $image_width := $size.width -}} {{- $image_height := $size.height -}} {{/* Process if not a SVG image */}} {{- if not $cloudinary.is_svg }} {{/* Return desktop image, with the size of the transformed variant */}} {{- with $desktop_size -}} {{- $params := $cloudinary.base | append (partial "func/GetCloudinarySize.html" .) -}} {{- if partial "func/isCrop" . -}} {{- $params = $params | append "c_fill" | append $crop -}} {{- end -}} {{- $image_desktop = printf "%s%s/%s" $pathFile (delimit $params ",") $imageFile -}} {{/* Desktop retina: a dpr_2.0 variant; the format stays negotiated by f_auto. */}} {{- if $is_desktop_retina -}} {{- $image_desktop_2x = printf "%s%s/%s" $pathFile (print "dpr_2.0," (delimit $params ",")) $imageFile -}} {{- end -}} {{- $size = partial "func/GetRemoteImageSize.html" (dict "url" (print $pathFile (delimit $params ",") ",fl_getinfo/" $imageFile) "provider" "cloudinary" "node" "output" "strict" true) -}} {{- with $size.width }}{{ $image_width = . }}{{ end -}} {{- with $size.height }}{{ $image_height = . }}{{ end -}} {{- end -}} {{/* Return mobile image, 1x/2x/3x as dpr variants */}} {{- with $mobile_size -}} {{- $params := $cloudinary.base -}} {{- if partial "func/isCrop" . -}} {{- $params = $params | append "c_fill" | append $crop -}} {{- end -}} {{- $params = $params | append (partial "func/GetCloudinarySize.html" .) -}} {{- $image_mobile = printf "%s%s/%s" $pathFile (delimit $params ",") $imageFile -}} {{- $image_mobile_2x = printf "%s%s/%s" $pathFile (print "dpr_2.0," (delimit $params ",")) $imageFile -}} {{- $image_mobile_3x = printf "%s%s/%s" $pathFile (print "dpr_3.0," (delimit $params ",")) $imageFile -}} {{- end -}} {{- end -}} {{- return dict "image_desktop" $image_desktop "image_desktop_2x" $image_desktop_2x "image_mobile" $image_mobile "image_mobile_2x" $image_mobile_2x "image_mobile_3x" $image_mobile_3x "image_width" $image_width "image_height" $image_height -}}
