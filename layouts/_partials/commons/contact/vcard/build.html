{{- /* Wraps vCard properties into a `.vcf` resource named after the page slug and published next to the page, e.g. /team/jane-doe/jane-doe.vcf Callers build the properties of their own content type — see `persons/vcard.html` and `places/vcard.html` — and escape each value with `func/EscapeVCard.html`. The envelope, a missing `N`, the `URL` property and the target path are handled here. @param {page} page The page the vCard belongs to. @param {slice} properties The vCard properties, e.g. "FN:Jane Doe". @return {resource} The vCard resource. @example {{ $properties := slice (printf "FN:%s" (partial "func/EscapeVCard.html" .Title)) }} {{ partial "commons/contact/vcard/build.html" (dict "page" . "properties" $properties) }} */ -}} {{- $page := .page -}} {{- $properties := .properties -}} {{- /* vCard 3.0 requires the structured name N beside FN (RFC 2426, section 5). A caller without one (a place, a person with no first and last names) gets its formatted name as the family name, which address books then show as the name of the card. FN is already escaped, and its escaping also suits an N component. */ -}} {{- $hasName := false -}} {{- $formatted := "" -}} {{- range $properties -}} {{- if or (hasPrefix . "N:") (hasPrefix . "N;") }}{{ $hasName = true }}{{ end -}} {{- if hasPrefix . "FN:" }}{{ $formatted = strings.TrimPrefix "FN:" . }}{{ end -}} {{- end -}} {{- $lines := slice "BEGIN:VCARD" "VERSION:3.0" -}} {{- if not $hasName -}} {{- $lines = $lines | append (printf "N:%s;;;;" $formatted) -}} {{- end -}} {{- $lines = $lines | append $properties -}} {{- $lines = $lines | append (printf "URL:%s" $page.Permalink) -}} {{- $lines = $lines | append "END:VCARD" -}} {{- /* Publish next to the page, without the baseURL sub-path, and name the file after the page slug so the download carries the person or place name. Falls back to `contact-info` when the page has no slug (home page). */ -}} {{- $base := (urls.Parse site.BaseURL).Path | strings.TrimSuffix "/" -}} {{- $dir := strings.TrimPrefix $base $page.RelPermalink -}} {{- $name := path.Base $dir -}} {{- if in (slice "/" ".") $name -}} {{- $name = "contact-info" -}} {{- end -}} {{- $target := printf "%s%s.vcf" $dir $name -}} {{- return resources.FromString $target (printf "%s\r\n" (delimit $lines "\r\n")) -}}
