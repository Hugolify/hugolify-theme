{{/* Lists the terms of a taxonomy as filters: the pinned ones first, then by title. Persons come as on the persons page. - tax (string) required, the taxonomy name - current (page) the page being rendered, its term is flagged active - limit (int) terms shown before the others are folded behind a "show more", site.Params.filters.more.limit or 5 by default. Only read where the fold is asked for, site.Params.filters.more.enable being false by default: the whole list is rendered otherwise. - mini_card (boolean) shows the avatar of each term - nested (boolean) lists the terms picked in a nested collection, whose key holds the path of their page ("parent/child", a sub-expertise). True by default, false keeps the top level only. */}} {{- $current := .current | default false }} {{/* Compared by URL: a term stands for another page when it has one (see func/GetRealContext), and a term key ending with a stray space comes back as another page object with the same URL. */}} {{- $currentURL := "" }} {{- with $current }}{{ $currentURL = .RelPermalink | default "" }}{{ end }} {{- $tax := .tax | default false }} {{- $isMiniCard := .mini_card | default false }} {{- $nested := true }} {{- if isset . "nested" }}{{ $nested = .nested }}{{ end }} {{- $isTags := strings.Contains $tax "_tags" }} {{/* The fold is a
dressed by .filters-more, which comes with hugolify-theme-design-system, hence the param rather than a plain count. filters.limit is where the count used to live, read here so a site that still declares it keeps its own; goes away in the next major. */}} {{- $isMore := site.Params.filters.more.enable | default false }} {{- $limit := .limit | default site.Params.filters.more.limit | default 5 }} {{- if $tax }} {{- $the_tax := index site.Taxonomies $tax }} {{- with $the_tax }} {{- $nTaxs := $the_tax | len }} {{- $i18n := printf "%s.title" $tax }} {{- $terms := slice }} {{- $persons := slice }} {{- range $key, $entry := . }} {{- if or $nested (not (strings.Contains $key "/")) }} {{/* The term page is looked up by its key: .Page of the entry is the term of the first page listed, and a page holding both "parent" and "parent/child" hands back "parent/child" for both (Hugo 0.165). */}} {{- $termPage := site.GetPage (printf "/%s/%s" $tax $key) | default $entry.Page }} {{- $page := partial "func/GetRealContext" $termPage -}} {{/* Sorted on the title item.html shows: a term with no content of its own shows its key, humanized */}} {{- $title := $page.Title }} {{- if not $page.File }}{{ $title = humanize $title }}{{ end }} {{/* The weight is read on the term itself (a category, a statute): the one of the page a term stands for only orders it among its siblings, like the position of a sub-expertise. */}} {{- $term := dict "page" $page "url" $termPage.Permalink "active" (and (ne $currentURL "") (or (eq $currentURL $termPage.RelPermalink) (eq $currentURL $page.RelPermalink))) "mini_card" $isMiniCard "key" (printf "%010d-%s" (cond (gt $termPage.Weight 0) $termPage.Weight 9999999999) $title) }} {{- if eq $page.Section "persons" }} {{- $persons = $persons | append $term }} {{- else }} {{- $terms = $terms | append $term }} {{- end }} {{- end }} {{- end }} {{/* Ranging over the taxonomy gives the terms by their key, byte by byte: a slug, not the title, and the accented ones last. sort compares in the collation of the language. */}} {{- $terms = sort $terms "key" }} {{/* Persons by statute, the pinned ones first, then by lastname */}} {{- with $persons }} {{- $pages := slice }} {{- range sort . "key" }} {{- $pages = $pages | append .page }} {{- end }} {{- $ranks := newScratch }} {{- range $rank, $person := partial "func/GetPersonsOrderByStatutes.html" $pages }} {{- $ranks.Set $person.Path $rank }} {{- end }} {{- $ranked := slice }} {{- range . }} {{- $ranked = $ranked | append (merge . (dict "key" (printf "%05d" ($ranks.Get .page.Path)))) }} {{- end }} {{- $terms = sort $ranked "key" | append $terms }} {{- end }} {{- $shown := $terms }} {{- $folded := slice }} {{- if $isMore }} {{- $shown = first $limit $terms }} {{- $folded = after $limit $terms }} {{- end }} {{- $title := i18n $i18n $nTaxs | default ($tax | humanize) }}
