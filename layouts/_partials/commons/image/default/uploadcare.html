{{/* Default image (fixed sizes, art direction) — Uploadcare. Used by commons/image.html when the responsive mode is off: no `sizes` nor `widths` given, and an eager image (lazy: false, e.g. the hero), a distinct mobile source (src_mobile) or a small image whose requested width is below image.responsive_min_width (320 by default). Each requested size (desktop, mobile) gives its own file, retina comes as 2x/3x density variants, the mobile one is served under (max-width: 576px). The srcset + sizes variant (browser picks the width) is in ../responsive/. Uploadcare: transformations on the fly; one desktop size, the mobile one in 1x/2x/3x; an SVG is served as-is. */}} {{/* Get params */}} {{- $desktop_size := .desktop_size -}} {{- $mobile_size := .mobile_size -}} {{- $mobile_size_2x := .mobile_size_2x -}} {{- $mobile_size_3x := .mobile_size_3x -}} {{- $crop := lower .crop -}} {{- $quality := printf "-/quality/%s/" (site.Params.image.quality | default "normal") -}} {{/* File name stripped for the transformations, SVG told by its extension */}} {{- $uploadcare := partial "func/GetUploadcareFile.html" .file -}} {{- $file := $uploadcare.file -}} {{- $is_svg := $uploadcare.is_svg -}} {{- $file_mobile := .file_mobile -}} {{- with $file_mobile -}} {{- $file_mobile = (partial "func/GetUploadcareFile.html" .).file -}} {{- end -}} {{/* Return params */}} {{- $image_desktop := $file -}} {{- $image_mobile := false -}} {{- $image_mobile_2x := false -}} {{- $image_mobile_3x := false -}} {{/* Intrinsic size. An URL without file name (/UUID/) has no extension: the format reported by Uploadcare tells */}} {{- $size := partial "func/GetRemoteImageSize.html" (dict "url" (printf "%s-/json/" $file) "provider" "uploadcare" "strict" true) -}} {{- $image_width := $size.width -}} {{- $image_height := $size.height -}} {{- $is_svg = or $is_svg (eq $size.format "svg") -}} {{/* Process if not a SVG image */}} {{- if not $is_svg }} {{/* Return desktop image, with the size of the resized variant */}} {{- with $desktop_size -}} {{- $image_desktop = printf "%s%s" $file (partial "func/GetUploadcareResize.html" (dict "size" . "crop" $crop "quality" $quality)) -}} {{- $size = partial "func/GetRemoteImageSize.html" (dict "url" (printf "%s-/resize/%s/-/json/" $file .) "provider" "uploadcare" "strict" true) -}} {{- with $size.width }}{{ $image_width = . }}{{ end -}} {{- with $size.height }}{{ $image_height = . }}{{ end -}} {{- end -}} {{/* Return mobile image, 1x/2x/3x */}} {{- if $mobile_size -}} {{- $image_mobile = printf "%s%s" $file_mobile (partial "func/GetUploadcareResize.html" (dict "size" $mobile_size "crop" $crop "quality" $quality)) -}} {{- $image_mobile_2x = printf "%s%s" $file_mobile (partial "func/GetUploadcareResize.html" (dict "size" $mobile_size_2x "crop" $crop "quality" $quality)) -}} {{- $image_mobile_3x = printf "%s%s" $file_mobile (partial "func/GetUploadcareResize.html" (dict "size" $mobile_size_3x "crop" $crop "quality" $quality)) -}} {{- end -}} {{- end -}} {{- return dict "image_desktop" $image_desktop "image_mobile" $image_mobile "image_mobile_2x" $image_mobile_2x "image_mobile_3x" $image_mobile_3x "image_width" $image_width "image_height" $image_height -}}
