{{/* Responsive image (srcset width descriptors) — Cloudinary. Used by commons/image.html when the caller gives `sizes` or `widths`, and by default for a lazy image: one srcset of widths (w descriptors) plus `sizes` ("auto" when lazy), the browser picks the file for the rendered width and the screen density. The fixed-size variant (eager, src_mobile, small images) is in ../default/. Transformations are generated on the fly (no build cost), so a dense ladder is fine. c_limit prevents upscaling; retina is handled by the browser via DPR. */}} {{- $file := .file -}} {{/* crop_ratio (h/w) enables cropped variants. */}} {{- $crop_ratio := .crop_ratio -}} {{/* /upload//, f_auto + quality, crop gravity */}} {{- $cloudinary := partial "func/GetCloudinaryParams.html" (dict "file" $file "crop" .crop) -}} {{- $pathFile := $cloudinary.path -}} {{- $imageFile := $cloudinary.file -}} {{- $crop := $cloudinary.gravity -}} {{- $image_desktop := $file -}} {{- $image_width := false -}} {{- $image_height := false -}} {{- $srcset := slice -}} {{- if not $cloudinary.is_svg -}} {{/* Intrinsic size: caps the ladder and feeds the aspect ratio. */}} {{- $size := partial "func/GetRemoteImageSize.html" (dict "url" (print $pathFile "fl_getinfo/" $imageFile) "provider" "cloudinary") -}} {{- $image_width = $size.width -}} {{- $image_height = $size.height -}} {{/* f_auto: Cloudinary negotiates AVIF/WebP/original per browser via the Accept header, so a single srcset suffices — no hand-built webp/avif needed. */}} {{- $base := $cloudinary.base -}} {{/* The ladder, capped by the source width (see func/GetSrcsetSizes). When the intrinsic size is unknown, every width is kept (c_limit / c_fill avoid upscaling server-side). */}} {{- $sizes := partial "func/GetSrcsetSizes.html" (dict "widths" .widths "max" $image_width "crop_ratio" $crop_ratio "provider" "cloudinary") -}} {{- $fallback := false -}} {{- $fallback_w := false -}} {{- $fallback_h := false -}} {{- range $sizes -}} {{- $w := .width -}} {{- $h := .height -}} {{- $p := $base -}} {{- if $crop_ratio -}} {{/* Crop: fill to w×h with the chosen gravity. */}} {{- $p = $p | append (printf "w_%d,h_%d,c_fill,%s" $w $h $crop) -}} {{- $fallback_w = $w -}} {{- $fallback_h = $h -}} {{- else -}} {{- $p = $p | append (printf "w_%d,c_limit" $w) -}} {{- end -}} {{- $srcset = $srcset | append (printf "%s%s/%s %dw" $pathFile (delimit $p ",") $imageFile $w) -}} {{- $fallback = printf "%s%s/%s" $pathFile (delimit $p ",") $imageFile -}} {{- end -}} {{/* Not `default`: it takes a boolean, false included, as a set value */}} {{- with $fallback -}} {{- $image_desktop = . -}} {{- end -}} {{/* For a crop, the dimensions are the fallback variant's computed crop size. */}} {{- if $crop_ratio -}} {{- $image_width = $fallback_w -}} {{- $image_height = $fallback_h -}} {{- end -}} {{- end -}} {{- return dict "image_desktop" $image_desktop "image_width" $image_width "image_height" $image_height "srcset" $srcset -}}
