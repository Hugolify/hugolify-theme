{{/* Responsive image (srcset width descriptors) — Hugo image processing. Used by commons/image.html when the caller gives `sizes` or `widths`, and by default for a lazy image: one srcset of widths (w descriptors) plus `sizes` ("auto" when lazy), the browser picks the file for the rendered width and the screen density. The fixed-size variant (eager, src_mobile, small images) is in ../default/. Retina is handled by the browser via DPR, so no explicit 2x/3x files are generated: the width ladder just needs to reach ~2x the largest render width. Variants larger than the source are skipped (no upscaling). AVIF is emitted since Hugo v0.162.0. */}} {{- $file := .file -}} {{- $filter := .filter -}} {{- $hint := .hint -}} {{- $image_desktop := false -}} {{- $image_width := false -}} {{- $image_height := false -}} {{- $srcset := slice -}} {{- $srcset_webp := slice -}} {{- $srcset_avif := slice -}} {{- $image := false -}} {{- with $file -}}{{- $image = resources.Get . -}}{{- end -}} {{- if $image -}} {{- if strings.HasSuffix $file "svg" -}} {{/* SVG is resolution-independent: serve as-is, no srcset. */}} {{- $dim := partial "func/GetImageDimensions.html" (dict "file" (print "/assets/" $file)) -}} {{- $image_desktop = $image.RelPermalink -}} {{- $image_width = index $dim 0 -}} {{- $image_height = index $dim 1 -}} {{- else -}} {{- $crop := .crop -}} {{- $crop_ratio := .crop_ratio -}} {{/* Crop mode (fill + anchor) keeps the requested aspect ratio at every width; otherwise plain resize. */}} {{- $proc := partial "func/GetHugoProcess.html" (dict "fill" $crop_ratio "crop" $crop "filter" $filter "hint" $hint) -}} {{/* AVIF encoding is available since Hugo v0.162.0; quality follows the imaging.avif config. */}} {{- $supports_avif := ge hugo.Version "0.162.0" -}} {{/* Widest variant the source can yield without upscaling: a crop is also bounded by the source height, otherwise Hugo stretches the crop box to the requested size and the image gets distorted. */}} {{- $max_width := $image.Width -}} {{- if $crop_ratio -}}{{- $max_width = int (math.Min $image.Width (math.Floor (div (float $image.Height) $crop_ratio))) -}}{{- end -}} {{/* The ladder, capped by $max_width (see func/GetSrcsetSizes) */}} {{- $sizes := partial "func/GetSrcsetSizes.html" (dict "widths" .widths "max" $max_width "crop_ratio" $crop_ratio "provider" "hugo") -}} {{- $fallback := false -}} {{- range $sizes -}} {{- $w := .width -}} {{/* Crop derives the height from the requested ratio; resize keeps the source ratio. */}} {{- $spec := printf "%dx" $w -}} {{- with .height -}}{{- $spec = printf "%dx%d" $w . -}}{{- end -}} {{- $r := $image.Process (printf "%s %s" $proc $spec) -}} {{- $rw := $image.Process (printf "%s %s webp" $proc $spec) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $w) -}} {{- $srcset_webp = $srcset_webp | append (printf "%s %dw" $rw.RelPermalink $w) -}} {{- if $supports_avif -}} {{- $ra := $image.Process (printf "%s %s avif" $proc $spec) -}} {{- $srcset_avif = $srcset_avif | append (printf "%s %dw" $ra.RelPermalink $w) -}} {{- end -}} {{- $fallback = $r -}} {{- end -}} {{/* Fallback : largest variant that fits, else one variant at the widest undistorted width. */}} {{- if not $fallback -}} {{- $sw := $max_width -}} {{- $spec := printf "%dx" $sw -}} {{- if $crop_ratio -}}{{- $spec = printf "%dx%d" $sw (int (math.Round (mul (float $sw) $crop_ratio))) -}}{{- end -}} {{- $fallback = $image.Process (printf "%s %s" $proc $spec) -}} {{- end -}} {{/* width/height follow the fallback variant, so a cropped ratio is reflected (no CLS). */}} {{- $image_desktop = $fallback.RelPermalink -}} {{- $image_width = $fallback.Width -}} {{- $image_height = $fallback.Height -}} {{- end -}} {{- end -}} {{- return dict "image_desktop" $image_desktop "image_width" $image_width "image_height" $image_height "srcset" $srcset "srcset_webp" $srcset_webp "srcset_avif" $srcset_avif -}}
